\section{Core Mathematical Expressions}

\subsection{Multi-Label Asymmetric Focal Loss ($\LASL$)}
Used for training high-capacity clinical deep neural networks under extreme positive-negative class imbalance:
\begin{equation}
\LASL = - \sum_{k=1}^{K} \left[ y_k (1 - p_k)^{\gamma_+} \log(p_k) + (1 - y_k) (p_{m,k})^{\gamma_-} \log(1 - p_{m,k}) \right]
\end{equation}
where $\gamma_+ = 1.0$ and $\gamma_- = 5.0$.

\subsection{Cosine Metaplasticity Annealing}
The time-varying learning rate $\eta_t$ during the critical training window satisfies:
\begin{equation}
\eta_t = \etaMin + \frac{1}{2}(\etaMax - \etaMin) \left[ 1 + \cos\left( \frac{\pi t}{T_{\max}} \right) \right]
\end{equation}

\subsection{Two-Compartment Pharmacokinetic Bolus Decay}
Plasma drug concentration $\Cp{t}$ following IV bolus administration:
\begin{equation}
\Cp{t} = \left[ \frac{D \cdot \ka}{\Vd \cdot (\ka - \ke)} \right] \left( e^{-\ke \cdot t} - e^{-\ka \cdot t} \right)
\end{equation}
